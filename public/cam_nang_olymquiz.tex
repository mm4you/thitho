\noindent\textbf{\Large TÓM TẮT CÁCH ĐIỀU HÀNH 1 TRẬN ĐẤU:}\\
\begin{enumerate}[leftmargin=*,itemsep=6pt]
    \item \textbf{Cấp mã số}: Vào trang quản lý thí sinh $\rightarrow$ Bấm copy link gửi cho 4 bạn kèm mã số PIN (ví dụ: \texttt{6AT7, 38T2...}).
    \item \textbf{Vào trận đấu}: Bật màn hình máy chiếu lên hội trường $\rightarrow$ Ban Giám Khảo mở Bàn Điều Hành.
    \item \textbf{Bấm 3 nút vàng}:
    \begin{itemize}[itemsep=2pt]
        \item Bấm nút \textbf{[Bắt Đầu]} (hoặc phím \textbf{Space}) để đồng hồ đếm ngược 15 giây.
        \item Hết giờ: Bấm nút \textbf{[Mở Đáp Án]} để máy chiếu hiện đáp án đúng.
        \item Bấm nút \textbf{[Chấm Điểm]} (hoặc phím \textbf{G}) để máy tự cộng điểm cho bạn nào làm đúng.
    \end{itemize}
    \item \textbf{Trao giải}: Hết 4 vòng thi, bấm nút \textbf{[Bảng Vàng Trao Giải]} để máy chiếu vinh danh Quán quân và bắn pháo hoa chúc mừng!
\end{enumerate}

\noindent{\Large\bfseries\color{primary} 1. TRANG CHỦ (KHI MỚI MỞ WEB LÊN)}\\[0.2cm]
\textbf{Nhìn vào hình dưới đây, bạn sẽ thấy 3 nút lớn tương ứng với 3 vai trò}:
\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item \textbf{Nút xanh dương [VÀO BỤC THI ĐẤU $\rightarrow$]}: Dành cho 4 bạn thí sinh bấm vào để vào phòng thi.
    \item \textbf{Nút màu vàng [MỞ MÀN MÁY CHIẾU]}: Dành cho Ban Tổ Chức chiếu lên màn hình lớn/máy chiếu cho cả hội trường xem.
    \item \textbf{Nút màu xanh lá [ĐĂNG NHẬP ĐIỀU HÀNH $\rightarrow$]}: Dành cho Ban Giám Khảo bấm vào để điều khiển trận đấu.
\end{itemize}

\noindent{\Large\bfseries\color{primary} 2. MÀN HÌNH CHỜ TRÊN MÁY CHIẾU HỘI TRƯỜNG}\\[0.2cm]

\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item Đây là màn hình trang trọng để chiếu lên máy chiếu trước giờ khai mạc hoặc lúc giải lao.
    \item Ban Giám Khảo có thể chuyển về màn hình này bất kỳ lúc nào bằng cách bấm nút \textbf{[Màn Hình Chờ]}.
    \item Nhấn phím \textbf{F11} trên bàn phím để mở toàn màn hình (không bị vướng thanh địa chỉ của trình duyệt).
\end{itemize}

\noindent{\Large\bfseries\color{primary} 3. MÀN HÌNH DÀNH CHO THÍ SINH CHỌN MÁY ĐẤU}\\[0.2cm]
\textbf{Thí sinh cần làm gì khi thấy màn hình này?}:
\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item \textbf{Bước 1}: Bấm chọn vị trí của mình (\textbf{MÁY 1, MÁY 2, MÁY 3 hoặc MÁY 4}).
    \item \textbf{Bước 2}: Nhập mã PIN do Ban Tổ Chức cấp vào ô trống.
    \item \textbf{Bước 3}: Điền Họ tên và Tên trường của mình (ví dụ: \textit{Nguyễn Văn A - THPT Chuyên Hà Nội}).
    \item \textbf{Bước 4}: Bấm nút màu vàng \textbf{[VÀO PHÒNG THI ĐẤU]} để vào bục máy sẵn sàng thi.
\end{itemize}

\noindent{\Large\bfseries\color{primary} 4. CỔNG ĐĂNG NHẬP DÀNH CHO BAN GIÁM KHẢO}\\[0.2cm]
\textbf{Cách Ban Giám Khảo đăng nhập}:
\begin{itemize}[leftmargin=*,itemsep=3pt]

    \item Bấm nút màu xanh dương \textbf{[VÀO BÀN ĐIỀU HÀNH TRẬN ĐẤU $\rightarrow$]}.
    \item Màn hình sẽ chuyển ngay sang Bàn Điều Hành Live để bấm giờ và chấm điểm.
\end{itemize}

% HÌNH 5: BÀN ĐIỀU HÀNH GIÁM KHẢO

\noindent{\Large\bfseries\color{primary} 5. BÀN ĐIỀU HÀNH BAN GIÁM KHẢO (QUAN TRỌNG NHẤT)}\\[0.2cm]
\textbf{Nhìn vào hình dưới, đây là nơi Giám Khảo điều khiển toàn bộ trận đấu}:
\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item \textbf{Hàng nút trên cùng}: Bấm để đổi màn hình máy chiếu (chiếu câu hỏi, giới thiệu thí sinh, chiếu luật thi, trao giải).
    \item \textbf{Đồng hồ bên trái}: Bấm nút vàng \textbf{[Bắt Đầu]} (hoặc phím \textbf{Space}) để chạy giờ. Bấm \textbf{[Reset]} để quay lại từ đầu.
    \item \textbf{Bộ 3 nút ở giữa}:
    \begin{itemize}[itemsep=2pt]
        \item \textbf{[Khóa Nộp]}: Khóa bục không cho thí sinh bấm nữa khi hết giờ.
        \item \textbf{[Mở Đáp Án]}: Chiếu đáp án đúng lên màn hình lớn cho cả hội trường xem.
        \item \textbf{[Chấm Điểm]}: Tự động cộng điểm vào bảng tổng cho bạn nào trả lời đúng.
    \end{itemize}
    \item \textbf{Cột bên phải}: Xem đáp án và số giây của 4 bạn. Có nút \textbf{+20, +30, -10} để cộng trừ điểm tay nếu cần.
\end{itemize}

\noindent{\Large\bfseries\color{primary} 6. TRANG QUẢN LÝ 4 THÍ SINH \& LẤY MÃ PIN}\\[0.2cm]
\textbf{Giám Khảo dùng trang này để chuẩn bị trước trận đấu}:
\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item \textbf{Nút [SAO CHÉP LINK THI ĐẤU]}: Bấm nút này ở từng máy để copy link và mã PIN (như \texttt{6AT7, 38T2...}) gửi cho 4 bạn thí sinh.
    \item \textbf{Nút [Reset Điểm Về 0] (góc trên phải)}: Đưa điểm của 4 bạn về 0 trước khi bắt đầu trận mới.
    \item \textbf{Nút [Tạo Mới 4 Mã PIN]}: Đổi mã PIN mới cho 4 máy nếu muốn bảo mật lại.
\end{itemize}

    \item Bấm nút xanh lá \textbf{[Import 1 File Cho Cả 4 Vòng]} nếu muốn nạp đề nhanh từ file văn bản.

\noindent{\Large\bfseries\color{primary} 8. HỘP THOẠI NẠP ĐỀ THI NHANH TRONG 1 GIÂY}\\[0.2cm]
\textbf{Cách nạp đề thi siêu nhanh không cần gõ từng câu}:
\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item \textbf{Cách 1}: Bấm nút \textbf{[Chọn File Từ Máy]} để tải file Word (\texttt{.docx}) hoặc file văn bản (\texttt{.txt}) lên.
    \item \textbf{Cách 2}: Copy toàn bộ đề thi rồi dán vào ô trống lớn ở giữa.
    \item Bấm nút xanh lá \textbf{[Nạp Vào Toàn Bộ 4 Vòng Thi]} $\rightarrow$ Hệ thống tự động chia đều câu hỏi vào 4 vòng ngay lập tức!

% ============================================================
\noindent{\Large\bfseries\color{primary} 11. MÀN HÌNH CHIẾU LUẬT THI CỦA 4 VÒNG ĐẤU}\\[0.2cm]
\textbf{Dùng để MC phổ biến thể lệ cho cả hội trường}:
\begin{itemize}[leftmargin=*,itemsep=3pt]
    \item \textbf{Vòng 1 (Khởi Động)}: 15 giây/câu, mỗi câu đúng được +10 điểm.
    \item \textbf{Vòng 2 (Vượt Chướng Ngại Vật)}: Trả lời 4 hàng ngang và đoán từ khóa chính bí mật.
    \item \textbf{Vòng 3 (Tăng Tốc)}: Cuộc đua phản xạ thời gian (10s, 20s, 30s, 40s) ai nhanh hơn được nhiều điểm hơn.
    \item \textbf{Vòng 4 (Về Đích)}: Gói câu hỏi về đích, có quyền đặt Ngôi Sao Hy Vọng để nhân đôi điểm số.
\end{itemize}

\noindent{\Large\bfseries\color{primary} 12. MÀN HÌNH BẢNG VÀNG TRAO GIẢI CHO QUÁN QUÂN}\\[0.2cm]

    \item Màn hình tự động xếp hạng: \textbf{QUÁN QUÂN -- Á QUÂN -- HẠNG 3 -- HẠNG 4} dựa trên tổng điểm.
    \item Tự động bắn pháo hoa giấy rực rỡ và phát nhạc chiến thắng để trao giải!
\end{itemize}
